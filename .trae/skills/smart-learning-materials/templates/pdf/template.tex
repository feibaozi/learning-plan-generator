\newcommand{\makecoverpage}[4]{%
  \thispagestyle{empty}
  \begin{tikzpicture}[remember picture, overlay]
    % Background gradient
    \shade[top color=primary!15, bottom color=accent!10]
      (current page.south west) rectangle (current page.north east);

    % Decorative shapes
    \fill[primary!8] (current page.north west) circle (6cm);
    \fill[accent!6] (current page.south east) circle (5cm);

    % Title block
    \node[align=center, text width=14cm] at (current page.center) {
      {\Huge\bfseries\textcolor{textDark}{#1}\par}
      \vspace{16pt}
      {\Large\textcolor{primary}{知识深度研究报告}\par}
      \vspace{32pt}
      \begin{tikzpicture}
        \node[fill=primary, inner sep=10pt, rounded corners=8pt] {
          \textcolor{white}{\large\bfseries #2}
        };
      \end{tikzpicture}
      \par\vspace{24pt}
      {\normalsize\textcolor{textMid}{领域：#3 \quad 深度：#4}\par}
      \vspace{8pt}
      {\small\textcolor{textLight}{由 Smart Learning Materials Generator 生成}\par}
    };
  \end{tikzpicture}
  \newpage
  \setcounter{page}{1}
}
